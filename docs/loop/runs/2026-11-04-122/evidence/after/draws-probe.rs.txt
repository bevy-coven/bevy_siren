// Iteration 122's probe, run again on the fixed tree (the same file as evidence/before/draws-on-the-unchanged-tree.txt).
// Deleted after its output was kept; this is the source as it ran after the fix.
use bevy::prelude::*;
use bevy_siren::*;

fn table() -> CueTable {
    CueTable::new(vec![
        Cue { sounds: vec![SoundSlot::new("a")], pitch: PitchRange::exact(0.0), repeat: Repeat::Shuffle, ..Cue::new("gun.burst") },
        Cue { sounds: vec![SoundSlot::new("a"), SoundSlot::new("b")], repeat: Repeat::AvoidRepeat { avoid: 1 }, ..Cue::new("footstep.grass") },
        // The same alternating cue with a pitch that cannot vary: its only decision is its FIRST pick.
        Cue { sounds: vec![SoundSlot::new("a"), SoundSlot::new("b")], pitch: PitchRange::exact(0.0), repeat: Repeat::AvoidRepeat { avoid: 1 }, ..Cue::new("footstep.exact") },
        Cue { sounds: vec![SoundSlot::new("a")], pitch: PitchRange::exact(0.0), repeat: Repeat::Shuffle, ..Cue::new("ui.click") },
        Cue { sounds: vec![SoundSlot::new("a"), SoundSlot::new("b"), SoundSlot::new("c")], repeat: Repeat::Shuffle, ..Cue::new("hit") },
        Cue {
            sounds: vec![SoundSlot::new("a"), SoundSlot::new("b"), SoundSlot::new("c")],
            repeat: Repeat::AvoidRepeat { avoid: 1 },
            ..Cue::new("footstep.stone")
        },
    ])
}

fn sounds() -> SirenSounds {
    let mut sounds = SirenSounds::new();
    for key in ["a", "b", "c"] {
        sounds.insert(key, Handle::<AudioSource>::default());
    }
    sounds
}

fn app() -> App {
    let mut app = App::new();
    app.add_plugins((MinimalPlugins, AssetPlugin::default()))
        .init_asset::<AudioSource>()
        .insert_resource(table())
        .insert_resource(sounds())
        .insert_resource(SirenSettings::default())
        .add_plugins(SirenPlugin);
    app.update();
    app
}

fn play(app: &mut App, cue: &str) -> Vec<Played> {
    app.world_mut().write_message(Play::cue(cue));
    app.update();
    app.world_mut().resource_mut::<Messages<Played>>().drain().collect()
}

fn main() {
    let mut alone = app();
    let alone_six: Vec<String> = (0..6).map(|_| play(&mut alone, "hit").remove(0).sound).collect();
    println!("observer alone (6 plays):            {alone_six:?}");

    for before in ["gun.burst", "footstep.grass", "footstep.exact", "ui.click"] {
        let mut app = app();
        for _ in 0..8 {
            let _ = play(&mut app, before);
        }
        let six: Vec<String> = (0..6).map(|_| play(&mut app, "hit").remove(0).sound).collect();
        println!("observer after 8 x {before:<15} {six:?}  {}", if six == alone_six { "SAME" } else { "DIFFERS" });
    }

    let mut one = app();
    let _ = play(&mut one, "footstep.grass");
    let one_six: Vec<String> = (0..6).map(|_| play(&mut one, "hit").remove(0).sound).collect();
    let mut eight = app();
    for _ in 0..8 {
        let _ = play(&mut eight, "footstep.grass");
    }
    let eight_six: Vec<String> = (0..6).map(|_| play(&mut eight, "hit").remove(0).sound).collect();
    println!("observer after 1 footstep:           {one_six:?}");
    println!("observer after 8 footsteps:          {eight_six:?}  {}", if eight_six == one_six { "SAME" } else { "DIFFERS" });

    // The alternating cue whose pitch cannot vary: one decision (its first pick) in eight plays, so eight plays must
    // leave the observer where ONE did.
    let mut exact_one = app();
    let _ = play(&mut exact_one, "footstep.exact");
    let exact_one_six: Vec<String> = (0..6).map(|_| play(&mut exact_one, "hit").remove(0).sound).collect();
    let mut exact_eight = app();
    for _ in 0..8 {
        let _ = play(&mut exact_eight, "footstep.exact");
    }
    let exact_eight_six: Vec<String> = (0..6).map(|_| play(&mut exact_eight, "hit").remove(0).sound).collect();
    println!("observer after 1 footstep.exact:     {exact_one_six:?}");
    println!("observer after 8 footsteps.exact:    {exact_eight_six:?}  {}", if exact_eight_six == exact_one_six { "SAME" } else { "DIFFERS" });

    let mut stone = app();
    let three: Vec<String> = (0..6).map(|_| play(&mut stone, "footstep.stone").remove(0).sound).collect();
    let repeat = three.windows(2).any(|w| w[0] == w[1]);
    println!("footstep.stone (avoid 1, 3 variants): {three:?}  repeats: {repeat}");
}
