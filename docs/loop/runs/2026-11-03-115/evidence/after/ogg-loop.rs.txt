//! `docs/loop/runs/2026-11-03-115/evidence/after/ogg-loop.rs.txt`.
//!
//!     cargo run --release --features mixer --example scratch_ogg_loop -- <evidence dir> <kenney dir> [long.ogg]

#[cfg(not(feature = "mixer"))]
fn main() {
    eprintln!("scratch_ogg_loop needs --features mixer (it uses the decoder the crate plays with)");
}

#[cfg(feature = "mixer")]
#[path = "../src/ogg.rs"]
mod ogg;

#[cfg(feature = "mixer")]
fn main() {
    run::main();
}

#[cfg(feature = "mixer")]
mod run {
    use std::io::Cursor;
    use std::num::NonZero;
    use std::path::{Path, PathBuf};
    use std::sync::Arc;
    use std::time::{Duration, Instant};

    use rodio::Source;
    use rodio::buffer::SamplesBuffer;

    use super::ogg;

    /// Bevy's own decoder call (`bevy_audio-0.20.1/src/audio_source.rs`), over the same bytes.
    fn decoder(bytes: &Arc<[u8]>) -> rodio::Decoder<Cursor<Arc<[u8]>>> {
        rodio::Decoder::builder()
            .with_byte_len(bytes.len() as u64)
            .with_data(Cursor::new(bytes.clone()))
            .build()
            .expect("decodes")
    }

    fn db(x: f32) -> f32 {
        if x <= 0.0 { f32::NEG_INFINITY } else { 20.0 * x.log10() }
    }

    fn files(dir: &Path, extension: &str, out: &mut Vec<PathBuf>) {
        let mut entries: Vec<_> =
            std::fs::read_dir(dir).expect("a directory").flatten().map(|e| e.path()).collect();
        entries.sort();
        for path in entries {
            if path.is_dir() {
                files(&path, extension, out);
            } else if path.extension().and_then(|e| e.to_str()) == Some(extension) {
                out.push(path);
            }
        }
    }

    fn write_wav(path: &Path, channels: u16, rate: u32, samples: &[f32]) {
        let data_len = (samples.len() * 4) as u32;
        let mut out = Vec::with_capacity(44 + samples.len() * 4);
        out.extend_from_slice(b"RIFF");
        out.extend_from_slice(&(36 + data_len).to_le_bytes());
        out.extend_from_slice(b"WAVEfmt ");
        out.extend_from_slice(&16u32.to_le_bytes());
        out.extend_from_slice(&3u16.to_le_bytes()); // IEEE float
        out.extend_from_slice(&channels.to_le_bytes());
        out.extend_from_slice(&rate.to_le_bytes());
        out.extend_from_slice(&(rate * u32::from(channels) * 4).to_le_bytes());
        out.extend_from_slice(&(channels * 4).to_le_bytes());
        out.extend_from_slice(&32u16.to_le_bytes());
        out.extend_from_slice(b"data");
        out.extend_from_slice(&data_len.to_le_bytes());
        for s in samples {
            out.extend_from_slice(&s.to_le_bytes());
        }
        std::fs::write(path, out).expect("writes");
    }

    /// The first index after 0 where the next 64 samples equal the first 64: where a loop's second pass begins, in samples.
    fn period(samples: &[f32]) -> Option<usize> {
        (1..samples.len().saturating_sub(64)).find(|&i| samples[i..i + 64] == samples[..64])
    }

    pub fn main() {
        let args: Vec<String> = std::env::args().collect();
        let (evidence, kenney) = (PathBuf::from(&args[1]), PathBuf::from(&args[2]));
        let long = args.get(3).map(PathBuf::from);

        // ---- P1, P2, P3: the fixture, through Bevy's chain, four passes, before and after ----
        println!(
            "== The fixture: tests/fixtures/tone.ogg through Bevy's chain (decoder, repeat_infinite) =="
        );
        let tone: Arc<[u8]> = std::fs::read("tests/fixtures/tone.ogg").expect("fixture").into();
        let stream = ogg::inspect(&tone).expect("the fixture declares its length");
        let duration = ogg::loop_length(&tone).expect("a loop length");
        let (rate, channels) = (stream.sample_rate, usize::from(stream.channels));
        println!(
            "declared: {} frames at {} Hz, {} channels; loop_length = {:?}",
            stream.frames, rate, channels, duration
        );

        let once: Vec<f32> = decoder(&tone).collect();
        let decoded_frames = once.len() / channels;
        println!(
            "decoder returns: {decoded_frames} frames ({} samples), {} frames past the declared length = {:.2} ms",
            once.len(),
            decoded_frames as u64 - stream.frames,
            (decoded_frames as f64 - stream.frames as f64) / f64::from(rate) * 1e3
        );
        let declared_samples = stream.frames as usize * channels;
        let pad = &once[declared_samples..];
        let pad_peak = pad.iter().fold(0.0f32, |m, s| m.max(s.abs()));
        let body_peak = once[..declared_samples].iter().fold(0.0f32, |m, s| m.max(s.abs()));
        println!("padding peak {:.1} dBFS; the stream's own peak {:.1} dBFS", db(pad_peak), db(body_peak));

        let before: Vec<f32> = decoder(&tone).repeat_infinite().take(4 * once.len()).collect();
        let after_len = declared_samples;
        let after: Vec<f32> =
            decoder(&tone).take_duration(duration).repeat_infinite().take(4 * after_len).collect();
        println!(
            "before.len() {} (four passes would be {}); after.len() {} (four would be {}); first current_span_len of a fresh decoder: {:?}; of one that has produced a sample: {:?}",
            before.len(),
            4 * once.len(),
            after.len(),
            4 * after_len,
            decoder(&tone).current_span_len(),
            {
                let mut d = decoder(&tone);
                d.next();
                d.current_span_len()
            }
        );
        // ---- every format: does `repeat_infinite` of a freshly built decoder produce samples at all? ----
        println!(
            "\n== Does a loop of each format's decoder produce samples? (decoder.repeat_infinite(), three passes asked for) =="
        );
        for (name, path) in [
            ("tone.ogg", "tests/fixtures/tone.ogg"),
            ("tone-441.ogg", "tests/fixtures/tone-441.ogg"),
            ("tone.flac", "tests/fixtures/tone.flac"),
            ("tone.mp3", "tests/fixtures/tone.mp3"),
        ] {
            let bytes: Arc<[u8]> = std::fs::read(path).expect("fixture").into();
            let (once, span) = (decoder(&bytes).count(), decoder(&bytes).current_span_len());
            let looped = decoder(&bytes).repeat_infinite().take(3 * once).count();
            println!(
                "{name:<13} one pass {once:>7} samples; fresh decoder span {span:?}; loop produced {looped} of {}",
                3 * once
            );
        }
        let p_before = period(&before).expect("a repeat");
        let p_after = period(&after).expect("a repeat");
        println!(
            "loop period before: {} frames ({:.3} ms); after: {} frames ({:.3} ms); declared {}",
            p_before / channels,
            (p_before / channels) as f64 / f64::from(rate) * 1e3,
            p_after / channels,
            (p_after / channels) as f64 / f64::from(rate) * 1e3,
            stream.frames
        );
        // The wrap: the step between the last sample of a pass and the first of the next, channel 0, against the largest step inside the pass.
        let step = |x: &[f32], at: usize| (x[at] - x[at - channels]).abs();
        let inner = |x: &[f32], n: usize| {
            (1..n / channels).map(|i| (x[i * channels] - x[(i - 1) * channels]).abs()).fold(0.0f32, f32::max)
        };
        println!(
            "wrap step before {:.4} (largest step inside a pass {:.4}); after {:.4} (largest {:.4})",
            step(&before, p_before),
            inner(&before, p_before),
            step(&after, p_after),
            inner(&after, p_after)
        );
        let silent = |x: &[f32], at: usize| {
            let (mut a, mut b) = (at, at);
            while a > 0 && x[a - 1].abs() < 1e-3 {
                a -= 1;
            }
            while b < x.len() && x[b].abs() < 1e-3 {
                b += 1;
            }
            (b - a) / channels
        };
        println!(
            "near-silent run (< -60 dBFS) at the first wrap: before {} frames ({:.2} ms), after {} frames",
            silent(&before, p_before),
            silent(&before, p_before) as f64 / f64::from(rate) * 1e3,
            silent(&after, p_after)
        );
        write_wav(&evidence.join("loop-tone-before.wav"), channels as u16, rate, &before);
        write_wav(&evidence.join("loop-tone-after.wav"), channels as u16, rate, &after);

        // ---- P4: the duration formula, against rodio's own TakeDuration ----
        println!(
            "\n== P4: cells (rate x channels x frames) where TakeDuration emits other than the declared samples =="
        );
        let (mut cells, mut naive_wrong, mut exact_wrong, mut worst_naive) = (0usize, 0usize, 0usize, 0i64);
        for rate in [8_000u32, 11_025, 22_050, 32_000, 44_100, 48_000, 88_200, 96_000, 192_000] {
            for ch in [1u16, 2] {
                for frames in [1_000u64, 12_345, 24_000, 48_000, 100_001, 480_000] {
                    let source = || {
                        SamplesBuffer::new(
                            NonZero::new(ch).unwrap(),
                            NonZero::new(rate).unwrap(),
                            vec![0.5f32; (frames as usize + 576) * usize::from(ch)],
                        )
                    };
                    let want = frames as i64 * i64::from(ch);
                    let naive = source()
                        .take_duration(Duration::from_secs_f64(frames as f64 / f64::from(rate)))
                        .count() as i64;
                    let exact =
                        source().take_duration(ogg::take_duration(frames, rate, ch).unwrap()).count() as i64;
                    cells += 1;
                    naive_wrong += usize::from(naive != want);
                    exact_wrong += usize::from(exact != want);
                    if (naive - want).abs() > worst_naive.abs() {
                        worst_naive = naive - want;
                    }
                }
            }
        }
        println!(
            "{cells} cells: frames/rate wrong in {naive_wrong} (worst {worst_naive:+} samples); N*p+1 wrong in {exact_wrong}"
        );

        // ---- P2, P5, P8: the population ----
        println!("\n== The population: Kenney's CC0 game audio, every .ogg ==");
        let mut paths = Vec::new();
        files(&kenney, "ogg", &mut paths);
        let (mut total, mut understood, mut with_tail, mut small_tail, mut exact_after) = (0, 0, 0, 0, 0);
        let (mut tails, mut peaks): (Vec<i64>, Vec<f32>) = (Vec::new(), Vec::new());
        let mut worst: Option<(i64, PathBuf)> = None;
        let mut timings = Vec::new();
        for path in &paths {
            total += 1;
            let bytes: Arc<[u8]> = std::fs::read(path).expect("reads").into();
            let t = Instant::now();
            let reps = 200;
            let mut inspected = None;
            for _ in 0..reps {
                inspected = std::hint::black_box(ogg::inspect(std::hint::black_box(&bytes)));
            }
            timings.push((t.elapsed().as_nanos() as f64 / f64::from(reps), bytes.len()));
            let Some(s) = inspected else {
                println!("  NOT UNDERSTOOD: {}", path.display());
                continue;
            };
            understood += 1;
            let d = decoder(&bytes);
            let ch = usize::from(d.channels().get());
            let samples: Vec<f32> = d.collect();
            let decoded = (samples.len() / ch) as i64;
            let tail = decoded - s.frames as i64;
            if tail > 0 {
                with_tail += 1;
                small_tail += usize::from(tail <= 1_024);
                tails.push(tail);
                peaks.push(samples[s.frames as usize * ch..].iter().fold(0.0f32, |m, x| m.max(x.abs())));
                if worst.as_ref().is_none_or(|(t, _)| tail > *t) {
                    worst = Some((tail, path.clone()));
                }
            }
            let trimmed = decoder(&bytes).take_duration(ogg::loop_length(&bytes).unwrap()).count();
            exact_after += usize::from(trimmed == s.frames as usize * ch);
        }
        tails.sort_unstable();
        peaks.sort_by(|a, b| a.partial_cmp(b).unwrap());
        println!(
            "{total} Ogg files; the walk understood {understood}; decoder returns more than declared for {with_tail} ({:.1}%), {small_tail} of them by 1..=1024 frames",
            100.0 * with_tail as f64 / total as f64
        );
        if !tails.is_empty() {
            println!(
                "tail frames: min {} median {} max {}; padding peak dBFS: median {:.1} worst {:.1}",
                tails[0],
                tails[tails.len() / 2],
                tails[tails.len() - 1],
                db(peaks[peaks.len() / 2]),
                db(*peaks.last().unwrap())
            );
        }
        println!("after the trim: {exact_after} of {understood} decode to exactly the declared samples");
        timings.sort_by(|a, b| a.0.partial_cmp(&b.0).unwrap());
        let (med, max) = (timings[timings.len() / 2], timings.last().unwrap());
        println!(
            "inspect() per call: median {:.2} us ({} bytes); slowest {:.2} us ({} bytes)",
            med.0 / 1e3,
            med.1,
            max.0 / 1e3,
            max.1
        );

        // ---- the worst file, four passes, before and after ----
        if let Some((tail, path)) = &worst {
            let bytes: Arc<[u8]> = std::fs::read(path).expect("reads").into();
            let s = ogg::inspect(&bytes).unwrap();
            let d = decoder(&bytes);
            let (ch, r) = (usize::from(d.channels().get()), d.sample_rate().get());
            let once: Vec<f32> = d.collect();
            let before: Vec<f32> = decoder(&bytes).repeat_infinite().take(4 * once.len()).collect();
            let after: Vec<f32> = decoder(&bytes)
                .take_duration(ogg::loop_length(&bytes).unwrap())
                .repeat_infinite()
                .take(4 * s.frames as usize * ch)
                .collect();
            println!(
                "\n== The worst tail: {} ({} frames of padding = {:.1} ms, {} Hz, {} ch, declared {}) ==",
                path.file_name().unwrap().to_string_lossy(),
                tail,
                *tail as f64 / f64::from(r) * 1e3,
                r,
                ch,
                s.frames
            );
            println!(
                "period before {:?} frames; after {:?} frames",
                period(&before).map(|p| p / ch),
                period(&after).map(|p| p / ch)
            );
            write_wav(&evidence.join("loop-worst-before.wav"), ch as u16, r, &before);
            write_wav(&evidence.join("loop-worst-after.wav"), ch as u16, r, &after);
        }

        // ---- P6: a long file ----
        if let Some(long) = long {
            let bytes = std::fs::read(&long).expect("reads");
            let reps = 200;
            let t = Instant::now();
            let mut got = None;
            for _ in 0..reps {
                got = std::hint::black_box(ogg::inspect(std::hint::black_box(&bytes)));
            }
            let each = t.elapsed().as_nanos() as f64 / f64::from(reps);
            println!(
                "\n== A long file: {} ({:.2} MB) ==\nunderstood: {:?}; inspect() {:.1} us per call",
                long.display(),
                bytes.len() as f64 / 1e6,
                got,
                each / 1e3
            );
        }
    }
}
